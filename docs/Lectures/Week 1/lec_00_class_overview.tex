\begin{frame}{Expectations for final project}
\phantomsection\label{expectations-for-final-project}
\begin{itemize}
\item
  Research paper or thesis chapter that could result in a peer-reviewed
  publication
\item
  Focus on applied time series analysis (univariate or multivariate)
\item
  Short format similar to ``Report'' in \emph{Ecology} or ``Rapid
  Communication'' in \emph{CJFAS}

  \begin{itemize}
  \tightlist
  \item
    Max of 20 pages, inclusive of refs, tables, figs, etc
  \item
    12-pt font, double-spaced throughout
  \end{itemize}
\end{itemize}
\end{frame}
